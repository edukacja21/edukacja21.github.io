🔁 KROK 8 — Pętla gry

while zycia > 0 and ukryte.count("_") > 0:
    # pytamy o literę
    # sprawdzamy trafienie
    # aktualizujemy hasło

    print("\n" + "=" * 30)
  
    print(wisielec[6 - zycia])
    
    # Wypisz już użyte litery
    if uzyte_litery:
        print(f"Użyte litery: {', '.join(uzyte_litery)}")
    
    # Pobierz literę od gracza
    while True:
        litera = input("Podaj literę: ").lower()
        
        # Walidacja wejścia
        if len(litera) != 1:
            print("Podaj JEDNĄ literę!")
            continue
        if not litera.isalpha():
            print("To nie jest litera!")
            continue
        if litera in uzyte_litery:
            print("Ta litera już była!")
            continue
        break
    
    # Dodaj literę do listy użytych
    uzyte_litery.append(litera)
    
    # Sprawdź czy litera jest w haśle
    if litera in haslo:
        print(f"✓ Dobrze! Litera '{litera}' jest w haśle.")
        
        # Odkryj wszystkie wystąpienia litery
        trafienie = False
        for i in range(len(haslo)):
            if haslo[i] == litera:
                ukryte[i] = litera
                trafienie = True
    else:
        print(f"✗ Źle! Litery '{litera}' nie ma w haśle.")
        zycia -= 1
        print(f"Zostało żyć: {zycia}")
    
    # Pokaż aktualny stan hasła
    print("\nHasło: " + " ".join(ukryte))
    
    # Pokaż postęp
    odkryte = ukryte.count("_")
    print(f"Odgadnęto: {len(haslo) - odkryte}/{len(haslo)} liter")
